JFreeChart Integration with TensorFlow C API Java

This page covers using JFreeChart 1.5.4 to visualize TensorFlow C API Java machine learning results, including image grids with predicted labels and accuracy percentages.

Setup

// Add to pom.xml

    org.jfree
    jfreechart
    1.5.4

Core Concept: XYPlot with Image Annotations

JFreeChart doesn't have a built-in image grid component, but we can use XYPlot with XYImageAnnotation and XYTextAnnotation to create a custom grid displaying:

  • Test images (28x28 pixels scaled up)
  • Predicted labels underneath each image
  • Per-class accuracy percentage under the label

Implementation Example

1. Create the Chart Dataset

import org.jfree.data.xy.XYSeriesCollection;

// Create empty series collection (we use annotations, not data points)
XYSeriesCollection dataset = new XYSeriesCollection();

// Set up axes (hidden, just serving as a coordinate system)
NumberAxis domain = new NumberAxis("");
domain.setRange(0, COLS);      // 5 columns
domain.setVisible(false);

NumberAxis range = new NumberAxis("");
range.setRange(0, ROWS);       // 2 rows
range.setInverted(true);       // Row 0 at top
range.setVisible(false);

// Create plot with no renderer (we use annotations)
XYPlot plot = new XYPlot(dataset, domain, range,
        new XYLineAndShapeRenderer(false, false));
plot.setBackgroundPaint(Color.WHITE);
plot.setOutlineVisible(false);
plot.setDomainGridlinesVisible(false);
plot.setRangeGridlinesVisible(false);

2. Add Image Annotations per Class

// For each of the 10 classes (digits 0-9 or fashion items):
for (int c = 0; c < NUM_CLASSES; c++) {
    int row = c / COLS, col = c % COLS;
    double cx = col + 0.5;  // center x coordinate
    
    // Load and scale the test image
    BufferedImage img = toImage(testX, sampleIdx[c]);
    plot.addAnnotation(new XYImageAnnotation(cx, row + 0.40, img));
    
    // Predict the class and get confidence
    int pred = Metrics.argmax(sampleProbs, c * NUM_CLASSES, NUM_CLASSES);
    boolean correct = pred == c;
    
    // Add predicted label below the image
    XYTextAnnotation label = new XYTextAnnotation(
            Metrics.FASHION_CLASS_NAMES[pred], cx, row + 0.72);
    label.setFont(new Font(Font.SANS_SERIF, Font.BOLD, 18));
    label.setTextAnchor(TextAnchor.CENTER);
    label.setPaint(correct ? Color.DARK_GRAY : Color.RED);
    label.setBackgroundPaint(Color.WHITE);
    plot.addAnnotation(label);
    
    // Add accuracy percentage under the label
    XYTextAnnotation acc = new XYTextAnnotation(
            String.format("accuracy %.1f%%", classAcc[c] * 100), cx, row + 0.86);
    acc.setFont(new Font(Font.SANS_SERIF, Font.PLAIN, 14));
    acc.setTextAnchor(TextAnchor.CENTER);
    acc.setPaint(Color.NUT_BROWN);
    acc.setBackgroundPaint(Color.WHITE);
    plot.addAnnotation(acc);
}

3. Complete Chart Creation

// Build the JFreeChart
JFreeChart chart = new JFreeChart(String.format(
        "Fashion-MNIST CNN - overall accuracy %.2f%%", overallAcc * 100),
        JFreeChart.DEFAULT_TITLE_FONT, plot, false);
chart.setBackgroundPaint(Color.WHITE);

// Create the chart panel and frame
ChartPanel panel = new ChartPanel(chart);
panel.setPreferredSize(new Dimension(COLS * 220, ROWS * 290));

JFrame frame = new JFrame("Fashion-MNIST classifier");
frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
frame.add(panel);
frame.pack();
frame.setLocationRelativeTo(null);
frame.setVisible(true);

Full Example: MnistClassifier.showGrid()

// From MnistClassifier.java - complete method
private static void showGrid(float[] testX, int[] sampleIdx, float[] sampleProbs,
                              double[] classAcc, double overallAcc) {
    XYSeriesCollection dataset = new XYSeriesCollection();
    
    NumberAxis domain = new NumberAxis("");
    domain.setRange(0, COLS);
    domain.setVisible(false);
    
    NumberAxis range = new NumberAxis("");
    range.setRange(0, ROWS);
    range.setInverted(true);
    range.setVisible(false);
    
    XYPlot plot = new XYPlot(dataset, domain, range,
            new XYLineAndShapeRenderer(false, false));
    plot.setBackgroundPaint(Color.WHITE);
    plot.setOutlineVisible(false);
    plot.setDomainGridlinesVisible(false);
    plot.setRangeGridlinesVisible(false);
    
    for (int c = 0; c < NUM_CLASSES; c++) {
        int row = c / COLS, col = c % COLS;
        double cx = col + 0.5;
        
        BufferedImage img = toImage(testX, sampleIdx[c]);
        plot.addAnnotation(new XYImageAnnotation(cx, row + 0.40, img));
        
        int pred = Metrics.argmax(sampleProbs, c * NUM_CLASSES, NUM_CLASSES);
        boolean ok = pred == c;
        
        XYTextAnnotation label = new XYTextAnnotation(
                "\"" + DIGIT_NAMES[pred] + "\"", cx, row + 0.72);
        label.setFont(new Font(Font.SANS_SERIF, Font.BOLD, 18));
        label.setTextAnchor(TextAnchor.CENTER);
        label.setPaint(ok ? Color.GREEN : Color.RED);
        label.setBackgroundPaint(Color.WHITE);
        plot.addAnnotation(label);
        
        XYTextAnnotation acc = new XYTextAnnotation(
                String.format("accuracy %.1f%%", classAcc[c] * 100), cx, row + 0.86);
        acc.setFont(new Font(Font.SANS_SERIF, Font.PLAIN, 14));
        acc.setTextAnchor(TextAnchor.CENTER);
        acc.setPaint(Color.BLUE);
        acc.setBackgroundPaint(Color.WHITE);
        plot.addAnnotation(acc);
    }
    
    JFreeChart chart = new JFreeChart(String.format(
            "MNIST digit CNN - overall accuracy %.2f%%", overallAcc * 100),
            JFreeChart.DEFAULT_TITLE_FONT, plot, false);
    chart.setBackgroundPaint(Color.WHITE);
    
    ChartPanel panel = new ChartPanel(chart);
    panel.setPreferredSize(new Dimension(COLS * 220, ROWS * 290));
    
    JFrame frame = new JFrame("MNIST digit classifier");
    frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
    frame.add(panel);
    frame.pack();
    frame.setLocationRelativeTo(null);
    frame.setVisible(true);
}

Key JFreeChart 1.5.x API Notes

  • XYImageAnnotation(double x, double y, Image image) - draws image centered at data coordinates (x,y)
  • XYTextAnnotation(String text, double x, double y) - adds text at data coordinates
  • TextAnchor.CENTER - centers text at the coordinate point (JFreeChart 1.5.x package: org.jfree.chart.ui.TextAnchor)
  • NumberAxis.setInverted(true) - flips y-axis so row 0 appears at the top
  • XYLineAndShapeRenderer(false, false) - disables lines and shapes (we only use annotations)

Chart Output Example

The resulting window displays a 5×2 grid (5 columns, 2 rows = 10 total positions) showing:

  • One sample test image per class (digits 0-9 or fashion items)
  • The predicted label name in bold underneath each image
  • The per-class accuracy percentage in smaller text under the label
  • Green text = correct prediction, Red text = incorrect

Integration with TensorFlow Results

The chart is populated from the evaluation phase:

// After training and evaluating:
double[] classAcc = classAccuracy(conf);  // per-class accuracy from confusion matrix
int[] sampleIdx = firstIndexOfEachClass(testY);  // first test index per class
float[] samples = new float[NUM_CLASSES * IMG_H * IMG_W];
for (int c = 0; c < NUM_CLASSES; c++)
    System.arraycopy(testX, sampleIdx[c] * IMG_H * IMG_W,
            samples, c * IMG_H * IMG_W, IMG_H * IMG_W);
float[] sampleProbs = net.predict(samples);  // predict one sample per class

// Open the chart window
SwingUtilities.invokeLater(() -> showGrid(
        testX, sampleIdx, sampleProbs, classAcc, overallAcc));
